cmake_minimum_required(VERSION 3.25)
project(ssx_graphics_probe LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "This probe uses Windows DXGI and Direct3D 12.")
endif()
add_executable(ssx_graphics_probe main.cpp)
target_compile_features(ssx_graphics_probe PRIVATE cxx_std_17)
target_compile_definitions(ssx_graphics_probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(ssx_graphics_probe PRIVATE dxgi d3d12 user32)
